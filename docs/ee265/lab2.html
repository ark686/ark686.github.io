1. Introduction ​

In this lab, we will use MATLAB to examine some continuous signals, including:

  1. Unit step functions
  2. Unit ramp function

3. Appendix A - Mysubplot MATLAB Function ​

matlab
function Mysubplot(sp1, sp2, sp3, x, y, titlex, xdelta, ydelta)
    % sp1, sp2, sp3  subplot parameters
    % x, y           values to plot
    % titlex         text for title
    % xdelta         increment between x ticks
    % ydelta         increment between y ticks

    % Define subplot
    subplot(sp1, sp2, sp3);

    % Plot 'x' versus 'y'
    plot(x, y);

    % Determine extents
    minx = round(min(x*10))/10;
    maxx = round(max(x)*10)/10;
    miny = round(min(y)*10)/10;
    maxy = round(max(y)*10)/10;

    % Set margins (full extent of 'x', 25% above and below 'y')
    axis([minx maxx (miny - ydelta) (maxy + ydelta)]);

    % Set x ticks
    xticks([minx:xdelta:maxx]);
    yticks([(miny - ydelta):ydelta:(maxy + ydelta)]);

    % Display title
    title(titlex);

    % Turn on grid
    grid on;
end

4. Appendix B - periodic1 MATLAB Function ​

matlab
function f = periodic1(tarray)
    T = 4;
    for i = 1:length(tarray)
        % Get current time
        t = tarray(i);

        % Map time to one cycle
        if t < 0
            t = t + (ceil(-t/T) * T);
        elseif t > T
            t = t - (floor(t/T) * T);
        end

        % Set output for time
        if t >= 0 && t < 1
            f(i) = 2 * t;
        elseif t >= 3 && t < 4
            f(i) = 2 * (1 - (t - 3));
        else
            f(i) = 0;
        end
    end
end
